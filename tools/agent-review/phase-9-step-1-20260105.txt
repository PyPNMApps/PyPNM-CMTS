set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="${SCRIPT_DIR}"
VENV_DIR=".env"
MODE=""
UPDATE_TAG=""

usage() {
  cat <<'USAGE_EOF'
PyPNM-CMTS Installer

Usage:
  ./install.sh --development [--venv-dir PATH]
  ./install.sh --update-ga [TAG] [--venv-dir PATH]
  ./install.sh --update-hot-fix [TAG] [--venv-dir PATH]
  ./install.sh --help

Modes:
  --development       Create/use a local venv and install editable + dev extras.
  --update-ga [TAG]   Update to the specified GA tag (or latest GA tag if omitted).
  --update-hot-fix [TAG]
                      Update to the specified hot-fix tag (or latest hot-fix tag if omitted).

Options:
  --venv-dir PATH     Virtual environment directory (default: .env).
  --help, -h          Show this help and exit.

Examples:
  ./install.sh --development
  ./install.sh --development --venv-dir .env-dev
  ./install.sh --update-ga v0.1.39.0
  ./install.sh --update-hot-fix v0.1.39.1
USAGE_EOF
}

require_command() {
  local command_name="$1"
  if ! command -v "${command_name}" >/dev/null 2>&1; then
    echo "ERROR: ${command_name} not found in PATH." >&2
    exit 1
  fi
}

require_arg() {
  local option_name="$1"
  local option_value="${2:-}"
  if [[ "${option_value}" == "" || "${option_value}" == --* ]]; then
    echo "ERROR: ${option_name} requires a value." >&2
    usage
    exit 2
  fi
}

ensure_python_min_version() {
  python3 - <<'PYCODE'
import sys
if sys.version_info < (3, 10):
    raise SystemExit(f"ERROR: Python 3.10+ required; found {sys.version.split()[0]}")
PYCODE
}

resolve_tag_version() {
  local tag_value="$1"
  if [[ "${tag_value}" == v* ]]; then
    echo "${tag_value#v}"
  else
    echo "${tag_value}"
  fi
}

git_fetch_tags() {
  git -C "${PROJECT_ROOT}" fetch --tags >/dev/null 2>&1 || true
}

ensure_git_repo() {
  if [[ ! -d "${PROJECT_ROOT}/.git" ]]; then
    echo "ERROR: ${PROJECT_ROOT} is not a git repository; tag updates require a git clone." >&2
    exit 1
  fi
}

ensure_git_tag() {
  local tag_value="$1"
  ensure_git_repo
  git_fetch_tags
  if ! git -C "${PROJECT_ROOT}" rev-parse "${tag_value}" >/dev/null 2>&1; then
    echo "ERROR: Tag not found: ${tag_value}" >&2
    exit 1
  fi
}

is_ga_tag() {
  local tag_value="$1"
  local version_value
  version_value="$(resolve_tag_version "${tag_value}")"
  case "${version_value}" in
    *.*.*.0) return 0 ;;
    *)       return 1 ;;
  esac
}

is_hot_fix_tag() {
  local tag_value="$1"
  local version_value
  local last_segment
  version_value="$(resolve_tag_version "${tag_value}")"
  last_segment="${version_value##*.}"
  if [[ "${version_value}" != *.*.*.* ]]; then
    return 1
  fi
  if [[ "${last_segment}" =~ ^[0-9]+$ ]] && [[ "${last_segment}" -gt 0 ]]; then
    return 0
  fi
  return 1
}

select_latest_ga_tag() {
  local tag
  local latest=""
  ensure_git_repo
  git_fetch_tags
  while IFS= read -r tag; do
    if is_ga_tag "${tag}"; then
      latest="${tag}"
    fi
  done < <(git -C "${PROJECT_ROOT}" tag --list "v*" | sort -V)
  echo "${latest}"
}

select_latest_hot_fix_tag() {
  local tag
  local latest=""
  ensure_git_repo
  git_fetch_tags
  while IFS= read -r tag; do
    if is_hot_fix_tag "${tag}"; then
      latest="${tag}"
    fi
  done < <(git -C "${PROJECT_ROOT}" tag --list "v*" | sort -V)
  echo "${latest}"
}

ensure_venv() {
  if [[ ! -d "${VENV_DIR}" ]]; then
    python3 -m venv "${VENV_DIR}"
  fi
}

activate_venv() {
  # shellcheck source=/dev/null
  source "${VENV_DIR}/bin/activate"
}

install_development() {
  echo "✅ Installing PyPNM-CMTS in development mode"
  ensure_venv
  activate_venv
  python -m pip install --upgrade pip setuptools wheel
  python -m pip install -e "${PROJECT_ROOT}[dev]"
  python - <<'PYCODE'
import pypnm_cmts
print(f"Installed PyPNM-CMTS v{pypnm_cmts.__version__}")
PYCODE
  echo "Next steps:"
  echo "  source ${VENV_DIR}/bin/activate"
  echo "  pypnm-cmts serve --help"
}

install_from_tag_via_worktree() {
  local tag_value="$1"
  local label="$2"
  local version_value
  local worktree_dir

  if [[ "${tag_value}" == "" ]]; then
    if [[ "${label}" == "GA" ]]; then
      tag_value="$(select_latest_ga_tag)"
    else
      tag_value="$(select_latest_hot_fix_tag)"
    fi
  fi
  if [[ "${tag_value}" == "" ]]; then
    echo "ERROR: No matching tags found for ${label} update." >&2
    exit 1
  fi

  ensure_git_tag "${tag_value}"
  version_value="$(resolve_tag_version "${tag_value}")"

  worktree_dir="$(mktemp -d)"
  cleanup_worktree() {
    git -C "${PROJECT_ROOT}" worktree remove --force "${worktree_dir}" >/dev/null 2>&1 || true
    rm -rf "${worktree_dir}" >/dev/null 2>&1 || true
  }
  trap cleanup_worktree EXIT

  echo "✅ Updating to ${label} tag ${tag_value}"
  git -C "${PROJECT_ROOT}" worktree add --detach "${worktree_dir}" "${tag_value}" >/dev/null

  ensure_venv
  activate_venv
  python -m pip install --upgrade pip setuptools wheel
  python -m pip install "${worktree_dir}"

  python - <<PYCODE
import pypnm_cmts
version = pypnm_cmts.__version__
if version != "${version_value}":
    raise SystemExit(f"Installed version {version} does not match tag ${tag_value}")
print(f"Installed PyPNM-CMTS v{version}")
PYCODE

  echo "Next steps:"
  echo "  source ${VENV_DIR}/bin/activate"
  echo "  pypnm-cmts --help"
}

if [[ $# -eq 0 ]]; then
  usage
  exit 1
fi

while [[ $# -gt 0 ]]; do
  case "$1" in
    --development)
      MODE="development"
      shift
      ;;
    --update-ga)
      MODE="update-ga"
      UPDATE_TAG="${2:-}"
      if [[ "${UPDATE_TAG}" != "" && "${UPDATE_TAG}" != --* ]]; then
        shift 2
      else
        UPDATE_TAG=""
        shift
      fi
      ;;
    --update-hot-fix)
      MODE="update-hot-fix"
      UPDATE_TAG="${2:-}"
      if [[ "${UPDATE_TAG}" != "" && "${UPDATE_TAG}" != --* ]]; then
        shift 2
      else
        UPDATE_TAG=""
        shift
      fi
      ;;
    --venv-dir)
      require_arg "--venv-dir" "${2:-}"
      VENV_DIR="${2}"
      shift 2
      ;;
    --help|-h)
      usage
      exit 0
      ;;
    *)
      echo "ERROR: Unknown option: $1" >&2
      usage
      exit 2
      ;;
  esac
done

if [[ "${MODE}" == "" ]]; then
  echo "ERROR: Missing mode flag." >&2
  usage
  exit 2
fi

require_command python3
ensure_python_min_version

case "${MODE}" in
  development)
    install_development
    ;;
  update-ga)
    require_command git
    install_from_tag_via_worktree "${UPDATE_TAG}" "GA"
    ;;
  update-hot-fix)
    require_command git
    install_from_tag_via_worktree "${UPDATE_TAG}" "hot-fix"
    ;;
  *)
    echo "ERROR: Unsupported mode: ${MODE}" >&2
    usage
    exit 2
    ;;
esac
