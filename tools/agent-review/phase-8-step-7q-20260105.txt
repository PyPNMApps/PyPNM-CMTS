T = TypeVar("T")


def _run_asyncio(coro: Awaitable[T]) -> T:
    try:
        asyncio.get_running_loop()
    except RuntimeError:
        loop = asyncio.new_event_loop()
        try:
            asyncio.set_event_loop(loop)
            return loop.run_until_complete(coro)
        finally:
            asyncio.set_event_loop(None)
            loop.close()
    raise RuntimeError("heavy poller cannot be run from inside an existing asyncio loop")


class HeavyInventoryProvider(Protocol):
    """Provider interface for heavy refresh inventory collection."""

    def fetch_inventory(
        self,
        sg_id: ServiceGroupId,
        settings: CmtsOrchestratorSettings,
    ) -> SgwHeavyInventoryModel:
        """
        Return RF channel inventory and cable modem membership.
        """


class SnmpInventoryProvider:
    """Inventory provider using CMTS SNMP discovery services."""

    def __init__(self) -> None:
        self.logger = logging.getLogger(f"{self.__class__.__name__}")

    def fetch_inventory(
        self,
        sg_id: ServiceGroupId,
        settings: CmtsOrchestratorSettings,
    ) -> SgwHeavyInventoryModel:
        try:
            topology = self._fetch_topology(settings)
            ds_channels, us_channels, ds_ch_set_id, us_ch_set_id = self._extract_channels(topology, sg_id)
            ds_sc_qam, us_sc_qam, ds_ofdm, us_ofdma = self._fetch_channel_inventory(settings)
            ds_rf_channels = _build_ds_rf_channels(ds_channels, ds_sc_qam, ds_ofdm)
            us_rf_channels = _build_us_rf_channels(us_channels, us_sc_qam, us_ofdma)
            modems = self._fetch_cable_modems(sg_id, settings)
            return SgwHeavyInventoryModel(
                ds_ch_set_id=ds_ch_set_id,
                us_ch_set_id=us_ch_set_id,
                ds_channel_ids=list(ds_channels),
                us_channel_ids=list(us_channels),
                ds_rf_channels=ds_rf_channels,
                us_rf_channels=us_rf_channels,
                cable_modems=modems,
            )
        except Exception as exc:
            raise RuntimeError(f"Heavy refresh inventory failed: {exc}") from exc

    @staticmethod
    def _extract_channels(
        topology: list[CmtsServiceGroupTopologyModel],
        sg_id: ServiceGroupId,
    ) -> tuple[list[int], list[int], ChSetId, ChSetId]:
        ds_channels: list[int] = []
        us_channels: list[int] = []
        ds_ch_set_id = ChSetId(0)
        us_ch_set_id = ChSetId(0)
        for entry in topology:
            if int(entry.md_cm_sg_id) != int(sg_id):
                continue
            ds_channels.extend(int(channel) for channel in entry.ds_channels)
            us_channels.extend(int(channel) for channel in entry.us_channels)
            ds_ch_set_id = entry.ds_ch_set_id
            us_ch_set_id = entry.us_ch_set_id
            break
        return (ds_channels, us_channels, ds_ch_set_id, us_ch_set_id)

    @staticmethod
    def _fetch_topology(settings: CmtsOrchestratorSettings) -> list[CmtsServiceGroupTopologyModel]:
        try:
            return _run_asyncio(
                CmtsTopologyCollector.fetch_service_group_topology(
                    cmts_hostname=settings.adapter.hostname,
                    read_community=settings.adapter.community,
                    write_community=SnmpWriteCommunity(str(settings.adapter.write_community)),
                    port=int(settings.adapter.port),
                )
            )[0]
        except Exception as exc:
            raise RuntimeError(f"Failed to collect service group topology: {exc}") from exc

    @staticmethod
    def _fetch_cable_modems(
        sg_id: ServiceGroupId,
        settings: CmtsOrchestratorSettings,
    ) -> list[SgwCableModemModel]:
        service = CmtsInventoryDiscoveryService(
            cmts_hostname=settings.adapter.hostname,
            read_community=SnmpReadCommunity(str(settings.adapter.community)),
            write_community=SnmpWriteCommunity(str(settings.adapter.write_community)),
            port=int(settings.adapter.port),
        )
        try:
            per_sg = _run_asyncio(service.discover_registered_cms_by_sg([sg_id]))
        except Exception as exc:
            raise RuntimeError(f"Failed to collect cable modem membership: {exc}") from exc
        if not per_sg:
            return []
        modems = [
            SgwCableModemModel(
                mac=cm.mac,
                ipv4=cm.ipv4,
                ipv6=cm.ipv6,
                registration_status=cm.registration_status,
                ds_channel_set=cm.ds_channel_set,
                us_channel_set=cm.us_channel_set,
                ds_channel_ids=list(cm.ds_channel_ids),
                us_channel_ids=list(cm.us_channel_ids),
            )
            for cm in per_sg[0].cms
        ]
        return sorted(modems, key=lambda modem: (str(modem.mac), str(modem.ipv4), str(modem.ipv6)))

    @staticmethod
    def _fetch_channel_inventory(
        settings: CmtsOrchestratorSettings,
    ) -> tuple[
        list[DocsIfDownstreamChannelEntry],
        list[DocsIfUpstreamChannelEntry],
        DocsIf31CmtsDsOfdmChanRecord,
        DocsIf31CmtsUsOfdmaChanRecord,
    ]:
        try:
            return _run_asyncio(
                CmtsChannelInventoryCollector.fetch_inventory(
                    cmts_hostname=settings.adapter.hostname,
                    read_community=SnmpReadCommunity(str(settings.adapter.community)),
                    port=int(settings.adapter.port),
                )
            )
        except Exception as exc:
            raise RuntimeError(f"Failed to collect RF channel inventory: {exc}") from exc


def _normalize_channels(channels: Iterable[int]) -> list[int]:
    normalized: set[int] = set()
    for channel in channels:
        value = int(channel)
        if value > 0:
            normalized.add(value)
    return sorted(normalized)


def _build_ds_rf_channels(
    channel_ids: Iterable[int],
    ds_sc_qam: list[DocsIfDownstreamChannelEntry],
    ds_ofdm: DocsIf31CmtsDsOfdmChanRecord,
) -> list[SgwRfChannelModel]:
    sc_qam_map = {entry.docsIfDownChannelId: entry for entry in ds_sc_qam}
    ofdm_map = {entry.docsIf31CmtsDsOfdmChanChannelId: entry for entry in ds_ofdm.entries}
    resolved: list[SgwRfChannelModel] = []
    for channel_id in _normalize_channels(channel_ids):
        entry = sc_qam_map.get(channel_id)
        if entry is not None:
            resolved.append(_build_sc_qam_channel(entry, RfChannelType.SC_QAM))
            continue
        ofdm_entry = ofdm_map.get(channel_id)
        if ofdm_entry is not None:
            resolved.append(_build_ofdm_channel(ofdm_entry))
    return resolved


def _build_us_rf_channels(
    channel_ids: Iterable[int],
    us_sc_qam: list[DocsIfUpstreamChannelEntry],
    us_ofdma: DocsIf31CmtsUsOfdmaChanRecord,
) -> list[SgwRfChannelModel]:
    sc_qam_map = {entry.docsIfUpChannelId: entry for entry in us_sc_qam}
    ofdma_map = {entry.docsIf31CmtsUsOfdmaChanId: entry for entry in us_ofdma.entries}
    resolved: list[SgwRfChannelModel] = []
    for channel_id in _normalize_channels(channel_ids):
        entry = sc_qam_map.get(channel_id)
        if entry is not None:
            resolved.append(_build_sc_qam_channel(entry, RfChannelType.SC_QAM))
            continue
        ofdma_entry = ofdma_map.get(channel_id)
        if ofdma_entry is not None:
            resolved.append(_build_ofdma_channel(ofdma_entry))
    return resolved


def _build_sc_qam_channel(
    entry: DocsIfDownstreamChannelEntry | DocsIfUpstreamChannelEntry,
    channel_type: RfChannelType,
) -> SgwRfChannelModel:
    center_frequency_hz = entry.frequency_hz
    channel_width_hz = entry.channel_width_hz
    lower_frequency_hz = _calculate_sc_qam_lower_frequency(center_frequency_hz, channel_width_hz)
    upper_frequency_hz = _calculate_sc_qam_upper_frequency(center_frequency_hz, channel_width_hz)
    return SgwRfChannelModel(
        channel_id=entry.channel_id,
        channel_type=channel_type,
        center_frequency_hz=center_frequency_hz,
        start_frequency_hz=None,
        plc_frequency_hz=None,
        channel_width_hz=channel_width_hz,
        lower_frequency_hz=lower_frequency_hz,
        upper_frequency_hz=upper_frequency_hz,
    )


def _build_ofdm_channel(entry: DocsIf31CmtsDsOfdmChanEntry) -> SgwRfChannelModel:
    lower_frequency_hz, upper_frequency_hz = _calculate_ofdm_bounds(entry)
    channel_width_hz = _calculate_channel_width(lower_frequency_hz, upper_frequency_hz)
    return SgwRfChannelModel(
        channel_id=entry.docsIf31CmtsDsOfdmChanChannelId,
        channel_type=RfChannelType.OFDM,
        center_frequency_hz=None,
        start_frequency_hz=None,
        plc_frequency_hz=entry.docsIf31CmtsDsOfdmChanPlcFreq,
        channel_width_hz=channel_width_hz,
        lower_frequency_hz=lower_frequency_hz,
        upper_frequency_hz=upper_frequency_hz,
    )


def _build_ofdma_channel(entry: DocsIf31CmtsUsOfdmaChanEntry) -> SgwRfChannelModel:
    lower_frequency_hz = entry.docsIf31CmtsUsOfdmaChanLowerBdryFreq
    upper_frequency_hz = entry.docsIf31CmtsUsOfdmaChanUpperBdryFreq
    channel_width_hz = _calculate_channel_width(lower_frequency_hz, upper_frequency_hz)
    start_frequency_hz = _calculate_ofdma_start_frequency(entry, lower_frequency_hz)
    return SgwRfChannelModel(
        channel_id=entry.docsIf31CmtsUsOfdmaChanId,
        channel_type=RfChannelType.OFDMA,
        center_frequency_hz=None,
        start_frequency_hz=start_frequency_hz,
        plc_frequency_hz=None,
        channel_width_hz=channel_width_hz,
        lower_frequency_hz=lower_frequency_hz,
        upper_frequency_hz=upper_frequency_hz,
    )


def _calculate_sc_qam_lower_frequency(
    center_frequency_hz: int | None,
    channel_width_hz: int | None,
) -> int | None:
    if center_frequency_hz is None or channel_width_hz is None:
        return None
    return int(center_frequency_hz - (channel_width_hz / 2))


def _calculate_sc_qam_upper_frequency(
    center_frequency_hz: int | None,
    channel_width_hz: int | None,
) -> int | None:
    if center_frequency_hz is None or channel_width_hz is None:
        return None
    return int(center_frequency_hz + (channel_width_hz / 2))


def _calculate_channel_width(lower_frequency_hz: int | None, upper_frequency_hz: int | None) -> int | None:
    if lower_frequency_hz is None or upper_frequency_hz is None:
        return None
    return int(upper_frequency_hz - lower_frequency_hz)


def _calculate_ofdm_bounds(
    entry: DocsIf31CmtsDsOfdmChanEntry,
) -> tuple[int | None, int | None]:
    zero = entry.docsIf31CmtsDsOfdmChanSubcarrierZeroFreq
    first = entry.docsIf31CmtsDsOfdmChanFirstActiveSubcarrierNum
    last = entry.docsIf31CmtsDsOfdmChanLastActiveSubcarrierNum
    spacing = entry.docsIf31CmtsDsOfdmChanSubcarrierSpacing
    lower_guard = entry.docsIf31CmtsDsOfdmChanLowerGuardbandWidth
    upper_guard = entry.docsIf31CmtsDsOfdmChanUpperGuardbandWidth
    if None in (zero, first, last, spacing, lower_guard, upper_guard):
        return (None, None)
    lower_frequency_hz = int(zero - ((first - 1) * spacing) - lower_guard)
    upper_frequency_hz = int((last * spacing) + zero + upper_guard)
    return (lower_frequency_hz, upper_frequency_hz)


def _calculate_ofdma_start_frequency(
    entry: DocsIf31CmtsUsOfdmaChanEntry,
    lower_frequency_hz: int | None,
) -> int | None:
    if lower_frequency_hz is None:
        return None
    zero = entry.docsIf31CmtsUsOfdmaChanSubcarrierZeroFreq
    spacing = entry.docsIf31CmtsUsOfdmaChanSubcarrierSpacing
    if zero is None or spacing is None:
        return None
    return int(lower_frequency_hz + (zero * spacing))


def _normalize_channel_summary(channels: Iterable[int]) -> SgwChannelSummaryModel:
    ids = _normalize_channels(channels)
    return SgwChannelSummaryModel(count=len(ids), channel_ids=ids)


def _summarize_modems(
    modems: Iterable[SgwCableModemModel],
) -> tuple[list[int], list[int], list[int], list[int]]:
    ds_set_counts: dict[int, int] = {}
    us_set_counts: dict[int, int] = {}
    ds_counts: dict[int, int] = {}
    us_counts: dict[int, int] = {}
    for modem in modems:
        ds_ids = [int(ch_id) for ch_id in modem.ds_channel_ids]
        us_ids = [int(ch_id) for ch_id in modem.us_channel_ids]
        for ch_id in ds_ids:
            ds_counts[ch_id] = ds_counts.get(ch_id, 0) + 1
        for ch_id in us_ids:
            us_counts[ch_id] = us_counts.get(ch_id, 0) + 1
        if modem.ds_channel_set is not None:
            ds_set_id = int(modem.ds_channel_set)
            ds_set_counts[ds_set_id] = ds_set_counts.get(ds_set_id, 0) + 1
        if modem.us_channel_set is not None:
            us_set_id = int(modem.us_channel_set)
            us_set_counts[us_set_id] = us_set_counts.get(us_set_id, 0) + 1
    return (
        _count_to_list(ds_set_counts),
        _count_to_list(us_set_counts),
        _count_to_list(ds_counts),
        _count_to_list(us_counts),
    )


def _count_to_list(counts: dict[int, int]) -> list[int]:
    return sorted(counts.items(), key=lambda item: item[0])


def _normalize_cable_modems(modems: Iterable[SgwCableModemModel]) -> list[SgwCableModemModel]:
    return sorted(modems, key=lambda modem: (str(modem.mac), str(modem.ipv4), str(modem.ipv6)))


def sgw_heavy_poller(
    sg_id: ServiceGroupId,
    settings: CmtsOrchestratorSettings,
    provider: HeavyInventoryProvider | None = None,
) -> SgwSnapshotPayloadModel:
    """
    Perform a heavy refresh for a single service group.
    """
    inventory_provider = provider if provider is not None else SnmpInventoryProvider()
    inventory = inventory_provider.fetch_inventory(sg_id, settings)
    modems = _normalize_cable_modems(inventory.cable_modems)
    ds_channel_set_counts, us_channel_set_counts, ds_channel_counts, us_channel_counts = _summarize_modems(modems)
    ds_sc_qam = [channel for channel in inventory.ds_rf_channels if channel.channel_type == RfChannelType.SC_QAM]
    ds_ofdm = [channel for channel in inventory.ds_rf_channels if channel.channel_type == RfChannelType.OFDM]
    us_sc_qam = [channel for channel in inventory.us_rf_channels if channel.channel_type == RfChannelType.SC_QAM]
    us_ofdma = [channel for channel in inventory.us_rf_channels if channel.channel_type == RfChannelType.OFDMA]
    return SgwSnapshotPayloadModel(
        ds_channels=_normalize_channel_summary(inventory.ds_channel_ids),
        us_channels=_normalize_channel_summary(inventory.us_channel_ids),
        cable_modems=modems,
        ds_rf_channels_sc_qam=ds_sc_qam,
        ds_rf_channels_ofdm=ds_ofdm,
        us_rf_channels_sc_qam=us_sc_qam,
        us_rf_channels_ofdma=us_ofdma,
        ds_channel_set_counts=ds_channel_set_counts,
        us_channel_set_counts=us_channel_set_counts,
        ds_channel_counts=ds_channel_counts,
        us_channel_counts=us_channel_counts,
    )
