from __future__ import annotations

import subprocess
import sys
import zipfile
from pathlib import Path

import pytest


DIST_DIR = Path("dist")
REPO_ROOT = Path(__file__).resolve().parents[1]
REQUIRED_ENTRIES = (
    "pypnm_cmts/settings/system.json",
    "pypnm_cmts/settings/cmts_system.json",
    "pypnm_cmts/settings/__init__.py",
)


def _find_latest_wheel(dist_dir: Path) -> Path:
    wheels = sorted(dist_dir.glob("*.whl"), key=lambda path: path.stat().st_mtime)
    if not wheels:
        raise AssertionError("No wheel artifacts found under dist/. Run build first.")
    return wheels[-1]


@pytest.mark.unit
def test_wheel_includes_settings_templates() -> None:
    subprocess.run(
        [sys.executable, "-m", "build"],
        cwd=REPO_ROOT,
        check=True,
    )

    wheel_path = _find_latest_wheel(DIST_DIR)
    with zipfile.ZipFile(wheel_path, "r") as wheel:
        names = set(wheel.namelist())

    for entry in REQUIRED_ENTRIES:
        assert entry in names
