from __future__ import annotations

import json
import subprocess
import sys
from pathlib import Path

import pytest


def _run_cli(args: list[str]) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [sys.executable, "-m", "pypnm_cmts.cli", *args],
        check=False,
        capture_output=True,
        text=True,
    )


@pytest.mark.unit
def test_config_init_creates_file(tmp_path: Path) -> None:
    config_path = tmp_path / "system.json"
    result = _run_cli(["config", "init", "--path", str(config_path)])

    assert result.returncode == 0
    assert config_path.exists()

    data = json.loads(config_path.read_text(encoding="utf-8"))
    assert "pypnm-cmts" in data
    assert "PnmFileRetrieval" in data


@pytest.mark.unit
def test_config_init_requires_force(tmp_path: Path) -> None:
    config_path = tmp_path / "system.json"
    config_path.write_text("{}", encoding="utf-8")
    result = _run_cli(["config", "init", "--path", str(config_path)])

    assert result.returncode == 1
    assert config_path.read_text(encoding="utf-8") == "{}"


@pytest.mark.unit
def test_config_validate_invalid(tmp_path: Path) -> None:
    config_path = tmp_path / "system.json"
    config_path.write_text("{}", encoding="utf-8")

    result = _run_cli(["config", "validate", "--path", str(config_path)])

    assert result.returncode == 2


@pytest.mark.unit
def test_config_validate_success(tmp_path: Path) -> None:
    config_path = tmp_path / "system.json"
    init_result = _run_cli(["config", "init", "--path", str(config_path)])
    assert init_result.returncode == 0

    data = json.loads(config_path.read_text(encoding="utf-8"))
    orchestrator = data.setdefault("CmtsOrchestrator", {})
    adapter = orchestrator.setdefault("adapter", {})
    adapter["hostname"] = "192.168.0.100"
    adapter["community"] = "public"
    config_path.write_text(f"{json.dumps(data, indent=2)}\n", encoding="utf-8")

    result = _run_cli(["config", "validate", "--path", str(config_path)])

    assert result.returncode == 0


@pytest.mark.unit
def test_config_show_pretty(tmp_path: Path) -> None:
    config_path = tmp_path / "system.json"
    init_result = _run_cli(["config", "init", "--path", str(config_path)])
    assert init_result.returncode == 0

    result = _run_cli(["config", "show", "--path", str(config_path), "--pretty"])

    assert result.returncode == 0
    payload = json.loads(result.stdout)
    assert "pypnm-cmts" in payload
